cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.json" XDZIP_VERSION_JSON)
string(JSON XDZIP_BASE_VERSION GET "${XDZIP_VERSION_JSON}" version)
string(JSON XDZIP_CHANNEL GET "${XDZIP_VERSION_JSON}" channel)
string(JSON XDZIP_BUILD_NUMBER GET "${XDZIP_VERSION_JSON}" build)
if(NOT XDZIP_BASE_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$" OR
   NOT XDZIP_BUILD_NUMBER MATCHES "^[1-9][0-9]*$" OR XDZIP_BUILD_NUMBER GREATER 65535)
    message(FATAL_ERROR "Invalid VERSION.json")
endif()
string(REPLACE "." ";" XDZIP_VERSION_COMPONENTS "${XDZIP_BASE_VERSION}")
foreach(component IN LISTS XDZIP_VERSION_COMPONENTS)
    if(component GREATER 65535 OR (component MATCHES "^0[0-9]"))
        message(FATAL_ERROR "Version components must fit Windows VERSIONINFO without leading zeros")
    endif()
endforeach()
set(XDZIP_FILE_FLAGS "0x0L")
set(XDZIP_DISPLAY_VERSION "${XDZIP_BASE_VERSION}")
if(NOT XDZIP_CHANNEL STREQUAL "stable")
    set(XDZIP_FILE_FLAGS "0x2L")
    set(XDZIP_SUFFIX "${XDZIP_CHANNEL}")
    if(XDZIP_CHANNEL STREQUAL "development")
        set(XDZIP_SUFFIX dev)
    elseif(NOT XDZIP_CHANNEL MATCHES "^(alpha|beta|rc)$")
        message(FATAL_ERROR "Invalid version channel")
    endif()
    string(APPEND XDZIP_DISPLAY_VERSION "-${XDZIP_SUFFIX}.${XDZIP_BUILD_NUMBER}")
endif()
project(xdzip_foundation VERSION ${XDZIP_BASE_VERSION} LANGUAGES CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.json")
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(XDZIP_BUILD_QT_PROBE "Build the independent Qt window probe" ON)
add_subdirectory(tools/bridge-probe)
if(XDZIP_BUILD_QT_PROBE)
    add_subdirectory(tools/toolchain-probe)
endif()

option(XDZIP_BUILD_DESKTOP_PREVIEW "Build the Qt component development preview" OFF)
if(XDZIP_BUILD_DESKTOP_PREVIEW)
    add_subdirectory(apps/desktop)
endif()
