cmake_minimum_required(VERSION 3.24)
project(xdzip_qt_network_security LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8.2 EXACT REQUIRED COMPONENTS Core Network Test)
add_executable(xdzip-qt-network-security main.cpp)
target_link_libraries(xdzip-qt-network-security PRIVATE Qt6::Core Qt6::Network ${CMAKE_DL_LIBS})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_library(xdzip-qt-network-abi OBJECT linux-abi-retain.cpp)
    target_link_libraries(xdzip-qt-network-abi PRIVATE Qt6::Core)
endif()
if(NOT IS_ABSOLUTE "${QTBASE_SOURCE}" OR NOT EXISTS "${QTBASE_SOURCE}/.tag")
    message(FATAL_ERROR "Use the verified patched QtBase source tree")
endif()
add_executable(xdzip-qt-authentication-tests
    "${QTBASE_SOURCE}/tests/auto/network/kernel/qauthenticator/tst_qauthenticator.cpp")
target_link_libraries(xdzip-qt-authentication-tests PRIVATE Qt6::CorePrivate Qt6::NetworkPrivate Qt6::Test)
if(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    target_compile_options(xdzip-qt-authentication-tests PRIVATE -include arm_acle.h)
endif()
