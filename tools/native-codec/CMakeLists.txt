foreach(pair IN ITEMS "COMMON_OBJS|CPP/Common" "WIN_OBJS|CPP/Windows" "7ZIP_COMMON_OBJS|CPP/7zip/Common" "AR_OBJS|CPP/7zip/Archive" "AR_COMMON_OBJS|CPP/7zip/Archive/Common" "7Z_OBJS|CPP/7zip/Archive/7z" "COMPRESS_OBJS|CPP/7zip/Compress" "CRYPTO_OBJS|CPP/7zip/Crypto" "C_OBJS|C")
  string(REPLACE "|" ";" fields "${pair}")
  list(GET fields 0 group)
  list(GET fields 1 directory)
  string(REGEX MATCH "(^|\n)${group} = [^\n]*(\n[ \t]+[^\n]+)*" block "${seven_manifest}")
  string(REGEX MATCHALL "[A-Za-z0-9_]+\\.obj" names "${block}")
  foreach(name IN LISTS names)
    string(REPLACE ".obj" "" stem "${name}")
    if(directory STREQUAL "C")
      list(APPEND seven_sources "${XDZIP_7ZIP_SOURCE}/${directory}/${stem}.c")
    else()
      list(APPEND seven_sources "${XDZIP_7ZIP_SOURCE}/${directory}/${stem}.cpp")
    endif()
  endforeach()
endforeach()

# Keep immutable downloads intact. These exact source transformations expose
# information hidden by the CLI-oriented API, without granting filesystem I/O.
set(seven_host "${CMAKE_CURRENT_BINARY_DIR}/seven-host/CPP/7zip/Archive")
foreach(format IN ITEMS Cab)
  file(MAKE_DIRECTORY "${seven_host}/${format}")
  file(GLOB handler_inputs "${XDZIP_7ZIP_SOURCE}/CPP/7zip/Archive/${format}/*.h" "${XDZIP_7ZIP_SOURCE}/CPP/7zip/Archive/${format}/*.cpp")
  foreach(input IN LISTS handler_inputs)
    get_filename_component(name "${input}" NAME)
    configure_file("${input}" "${seven_host}/${format}/${name}" COPYONLY)
  endforeach()
endforeach()
function(seven_replace format name old replacement)
  set(path "${seven_host}/${format}/${name}")
  file(READ "${path}" content)
  string(REPLACE "${old}" "" stripped "${content}")
  string(LENGTH "${content}" before)
  string(LENGTH "${stripped}" after)
  string(LENGTH "${old}" expected)
  math(EXPR removed "${before}-${after}")
  if(NOT removed EQUAL expected)
    message(FATAL_ERROR "7-Zip host boundary changed: ${format}/${name}")
  endif()
  string(REPLACE "${old}" "${replacement}" content "${content}")
  file(WRITE "${path}" "${content}")
endfunction()

# Replace UnRAR's file boundary, rather than permitting its ordinary filesystem
# implementation. The isolated copy preserves source delivery and avoids modifying
# the downloaded immutable input. Fail closed if its inline boundary changes.
set(unrar_copy "${CMAKE_CURRENT_BINARY_DIR}/unrar-host")
file(MAKE_DIRECTORY "${unrar_copy}")
file(GLOB unrar_inputs "${XDZIP_UNRAR_SOURCE}/*.hpp" "${XDZIP_UNRAR_SOURCE}/*.cpp")
foreach(input IN LISTS unrar_inputs)
  get_filename_component(name "${input}" NAME)
  if(NOT name STREQUAL "file.hpp")
    configure_file("${input}" "${unrar_copy}/${name}" COPYONLY)
  endif()
endforeach()
file(READ "${XDZIP_UNRAR_SOURCE}/file.hpp" file_header)
set(old_boundary "virtual bool IsOpened() {return hFile!=FILE_BAD_HANDLE;}")
string(FIND "${file_header}" "${old_boundary}" boundary_at)
if(boundary_at EQUAL -1)
  message(FATAL_ERROR "UnRAR File::IsOpened boundary changed; review the host-I/O adapter")
endif()
string(REPLACE "${old_boundary}" "virtual bool IsOpened();" file_header "${file_header}")
file(GENERATE OUTPUT "${unrar_copy}/file.hpp" CONTENT "${file_header}")
set(unrar_names rar strlist strfn pathfn smallfn global filefn filcreat archive arcread unicode system crypt crc rawread encname resource match timefn rdwrfn consio options errhnd rarvm secpassword rijndael getbits sha1 sha256 blake2s hash extinfo extract volume list find unpack headers threadpool rs16 cmddata ui largepage filestr scantree dll qopen)
