cmake_minimum_required(VERSION 3.24)
project(xdzip_msix_shell_probe LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "This fixture requires Windows")
endif()
get_filename_component(REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SHELL_SOURCE "${REPOSITORY}/apps/desktop/shell")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(PROJECT_SOURCE_DIR "${REPOSITORY}")
set(XDZIP_APP_ICON_ICO "${REPOSITORY}/design/assets/brand/xdzip.ico")
configure_file("${SHELL_SOURCE}/WindowsMenuIcons.rc.in" icons.rc @ONLY)
add_library(menu SHARED "${SHELL_SOURCE}/WindowsContextMenu.cpp"
    "${SHELL_SOURCE}/WindowsContextMenu.def" "${CMAKE_CURRENT_BINARY_DIR}/icons.rc")
set_target_properties(menu PROPERTIES OUTPUT_NAME xdzip-shell)
add_executable(probe WIN32 main.cpp fixture.rc)
target_link_options(probe PRIVATE /ENTRY:wmainCRTStartup)
set_target_properties(probe PROPERTIES OUTPUT_NAME xdzip)
target_include_directories(probe PRIVATE "${SHELL_SOURCE}")
add_executable(menu-tests "${SHELL_SOURCE}/WindowsMenuTest.cpp")
foreach(target menu probe menu-tests)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    target_link_libraries(${target} PRIVATE shell32 ole32 uuid advapi32 bcrypt version gdi32 user32 windowsapp)
endforeach()
